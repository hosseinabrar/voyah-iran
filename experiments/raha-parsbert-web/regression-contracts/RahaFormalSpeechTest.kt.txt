class RahaFormalSpeechTest {
 @Test fun formalVariantsKeepTheSameActionAndParameters() {
  val parser=RahaCommandRecognizer()
  val failures=mutableListOf<String>()
  var count=0
  for(base in examples) {
   val expected=parser.recognize(base)
   if(expected !is Recognition.Matched) { failures += "baseline: $base -> $expected"; continue }
   val formal=base.split(' ').joinToString(" ") { formalWords[it] ?: it }
   for(text in listOf(formal,"لطفاً $formal","خواهشمندم $formal")) {
    val actual=parser.recognize(text)
    if(actual !is Recognition.Matched || expected.command.action!=actual.command.action || expected.command.parameters!=actual.command.parameters)
      failures += "$text -> $actual; expected ${expected.command}"
    count++
   }
  }
  assertTrue(failures.take(60).joinToString("\n")+"\nFailures=${failures.size}",failures.isEmpty())
  println("Formal speech: $count variants passed")
 }
